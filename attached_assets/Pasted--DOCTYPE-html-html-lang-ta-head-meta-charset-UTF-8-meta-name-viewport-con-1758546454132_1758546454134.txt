    <script>
        function calculate() {
            // Get all inputs
            const cash1 = parseFloat(document.getElementById('cashCollected1').value) || 0;
            const cash2 = parseFloat(document.getElementById('cashCollected2').value) || 0;
            const outCash = parseFloat(document.getElementById('outCash').value) || 0;
            const op1 = parseFloat(document.getElementById('operatorAmount1').value) || 0;
            const op2 = parseFloat(document.getElementById('operatorAmount2').value) || 0;
            const outOperator = parseFloat(document.getElementById('outOperator').value) || 0;
            const passDeduction = parseFloat(document.getElementById('passDeduction').value) || 0;
            const startCNG = parseFloat(document.getElementById('startCNG').value) || 0;
            const endCNG = parseFloat(document.getElementById('endCNG').value) || 0;
            
            // CNG Rate is now a fixed value (hardcoded)
            const cngRate = 90;

            const totalCash = cash1 + cash2 + outCash;
            const inHouseOperatorTotal = op1 + op2;
            const grandTotalOperator = inHouseOperatorTotal + outOperator;

            if (grandTotalOperator > 0 || totalCash > 0 || startCNG > 0 || endCNG > 0 || passDeduction > 0) {
                document.getElementById('results-container').style.display = 'block';

                // Gross Salary Calculation
                let inHouseSalary = 0;
                if (inHouseOperatorTotal > 4500) {
                    inHouseSalary = (4500 * 0.30) + ((inHouseOperatorTotal - 4500) * 0.70);
                } else {
                    inHouseSalary = inHouseOperatorTotal * 0.30;
                }
                const outOperatorSalary = outOperator * 0.30;
                const grossSalary = inHouseSalary + outOperatorSalary;

                // Net Salary Calculation
                const insuranceDeduction = 60;
                const netSalary = grossSalary - insuranceDeduction;

                // CNG Calculation
                const baseCNG = grandTotalOperator * 0.30;
                const cngAdjustment = (startCNG - endCNG) * cngRate;
                const finalCNG = baseCNG - cngAdjustment;
                
                // Final Settlement Calculation
                const companySettlement = (totalCash - finalCNG) + passDeduction - netSalary;
                
                // Display results
                document.getElementById('totalCashValue').innerText = `₹${totalCash.toFixed(2)}`;
                document.getElementById('grandTotalOperatorValue').innerText = `₹${grandTotalOperator.toFixed(2)}`;
                
                document.getElementById('grossSalaryValue').innerText = `₹${grossSalary.toFixed(2)}`;
                document.getElementById('netSalaryValue').innerText = `₹${netSalary.toFixed(2)}`;
                
                const adjElement = document.getElementById('cngAdjustmentValue');
                document.getElementById('baseCNGValue').innerText = `₹${baseCNG.toFixed(2)}`;
                if (cngAdjustment > 0) {
                    adjElement.innerText = `- ₹${Math.abs(cngAdjustment).toFixed(2)}`;
                } else if (cngAdjustment < 0) {
                    adjElement.innerText = `+ ₹${Math.abs(cngAdjustment).toFixed(2)}`;
                } else {
                    adjElement.innerText = `₹0.00`;
                }
                document.getElementById('finalCNGValue').innerText = `₹${finalCNG.toFixed(2)}`;
                
                document.getElementById('companySettlementValue').innerText = `₹${companySettlement.toFixed(2)}`;
            } else {
                document.getElementById('results-container').style.display = 'none';
            }
        }
    </script>
